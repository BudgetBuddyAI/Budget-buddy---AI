<script>
let budget = Number(localStorage.getItem("budget")) || 0;
let expenses = JSON.parse(
  localStorage.getItem("expenses") || "[]"
);

const money = n => "₹" + n.toFixed(2);

function saveData() {
  localStorage.setItem("budget", budget);
  localStorage.setItem("expenses",
    JSON.stringify(expenses));
}

function saveBudget() {
  const value = Number(
    document.getElementById("budget").value
  );
  if (value <= 0) {
    alert("Please enter a valid budget.");
    return;
  }
  budget = value;
  saveData();
  render();
}

function addExpense() {
  const description =
    document.getElementById("description").value.trim();
  const amount =
    Number(document.getElementById("amount").value);
  const category =
    document.getElementById("category").value;

  if (!description || amount <= 0) {
    alert("Enter a description and valid amount.");
    return;
  }

  expenses.push({
    description, amount, category
  });

  document.getElementById("description").value = "";
  document.getElementById("amount").value = "";

  saveData();
  render();
}

function deleteExpense(index) {
  expenses.splice(index, 1);
  saveData();
  render();
}

function render() {
  const spent = expenses.reduce(
    (sum, item) => sum + item.amount, 0
  );
  const balance = budget - spent;

const usedPercent = budget > 0
  ? Math.min((spent / budget) * 100, 100)
  : 0;
const alertBox =
  document.getElementById("budgetAlert");

if (spent > budget && budget > 0) {
  alertBox.textContent =
    "🚨 Budget exceeded! Spending limit crossed.";
  alertBox.style.color = "#dc2626";

} else if (usedPercent >= 80 && budget > 0) {
  alertBox.textContent =
    "⚠️ Warning! You have used 80% of your budget.";
  alertBox.style.color = "#f59e0b";

} else {
  alertBox.textContent =
    "✅ Your budget is under control.";
  alertBox.style.color = "#16a34a";
}
document.getElementById("budgetPercent").textContent =
  usedPercent.toFixed(1) + "% Used";

document.getElementById("budgetBar").style.width =
  usedPercent + "%";

document.getElementById("budgetBar").style.background =
  usedPercent >= 100 ? "#dc2626" :
  usedPercent >= 80 ? "#f59e0b" : "#16a34a";

document.getElementById("budgetDetails").textContent =
  money(spent) + " spent / " + money(budget) + " budget";

  document.getElementById("totalBudget").textContent =
    money(budget);
  document.getElementById("totalSpent").textContent =
    money(spent);
  document.getElementById("remaining").textContent =
    money(balance);
  document.getElementById("count").textContent =
    expenses.length;

  const list = document.getElementById("expenseList");
  list.innerHTML = "";

  if (expenses.length === 0) {
    list.textContent = "No expenses yet.";
  }

  expenses.forEach((item, index) => {
    const row = document.createElement("div");
    row.className = "expense";

    const label = document.createElement("span");
    label.textContent =
      item.description + " · " + item.category +
      " · " + money(item.amount);

    const del = document.createElement("button");
    del.className = "delete";
    del.textContent = "Delete";
    del.onclick = () => deleteExpense(index);

    row.append(label, del);
    list.appendChild(row);
  });

const categoryTotals = {};
console.log("Expense categories:", expenses.map(item => item.category));
expenses.forEach(item => {
  categoryTotals[item.category] =
    (categoryTotals[item.category] || 0)
    + item.amount;
});

const summary =
  document.getElementById("categorySummary");

summary.innerHTML = "";

summary.innerHTML = "";

const totalCategorySpend = Object.values(categoryTotals)
  .reduce((total, amount) => total + amount, 0);

if (totalCategorySpend === 0) {
  summary.textContent = "No expenses yet.";
} else {
  for (const category in categoryTotals) {
    const amount = categoryTotals[category];
    const percent = (amount / totalCategorySpend) * 100;

    const container = document.createElement("div");
    container.style.marginBottom = "18px";

    const label = document.createElement("p");
    label.textContent =
      category + ": ₹" + amount.toFixed(2);

    const track = document.createElement("div");
    track.style.background = "#e5e7eb";
    track.style.height = "12px";
    track.style.borderRadius = "10px";
    track.style.overflow = "hidden";

    const bar = document.createElement("div");
    bar.style.width = percent + "%";
    bar.style.height = "100%";
    bar.style.background = "#2875e8";
    bar.style.borderRadius = "10px";

    track.appendChild(bar);
    container.appendChild(label);
    container.appendChild(track);
    summary.appendChild(container);
  }
}


  const insight = document.getElementById("insight");

  if (budget <= 0) {
    insight.textContent = "Please set your monthly budget.";
  } else if (balance < 0) {
    insight.textContent =
      "You have crossed your budget. Review your expenses.";
  } else if (spent / budget >= 0.8) {
    insight.textContent =
      "You have used 80% or more of your budget. Be careful with upcoming expenses.";
  } else {
    insight.textContent =
      "Good progress! Your spending is currently within budget.";
  }
}

document.getElementById("budget").value = budget || "";
render();
</script>
